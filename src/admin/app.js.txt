  /* ───────── helpers ───────── */
  function el(tag, attrs) {
    var n = document.createElement(tag);
    if (attrs) {
      Object.keys(attrs).forEach(function (k) {
        var v = attrs[k];
        if (v === null || v === undefined || v === false) return;
        if (k === "class") n.className = v;
        else if (k === "text") n.textContent = v;
        else if (k.slice(0, 2) === "on" && typeof v === "function") n.addEventListener(k.slice(2), v);
        else if (k === "value") n.value = v;
        else if (k === "checked") n.checked = !!v;
        else n.setAttribute(k, v === true ? "" : String(v));
      });
    }
    var kids = Array.prototype.slice.call(arguments, 2);
    (function add(list) {
      list.forEach(function (c) {
        if (c === null || c === undefined || c === false) return;
        if (Array.isArray(c)) return add(c);
        n.appendChild(c instanceof Node ? c : document.createTextNode(String(c)));
      });
    })(kids);
    return n;
  }
  function clear(n) { while (n.firstChild) n.removeChild(n.firstChild); return n; }
  function cur() { return (ME && ME.currency_symbol) || "$"; }
  function money(v) {
    return cur() + Number(v || 0).toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  }
  function fmtDate(iso) {
    if (!iso) return "—";
    var d = new Date(iso);
    if (isNaN(d.getTime())) return String(iso);
    return d.toLocaleString("pt-BR", { timeZone: "America/Sao_Paulo", day: "2-digit", month: "2-digit", year: "numeric", hour: "2-digit", minute: "2-digit" });
  }
  function fmtSize(n) {
    n = Number(n || 0);
    if (n >= 1048576) return (n / 1048576).toFixed(1) + " MB";
    if (n >= 1024) return Math.round(n / 1024) + " KB";
    return n + " B";
  }
  function userLabel(id, username) { return username ? "@" + username + " (" + id + ")" : String(id == null ? "—" : id); }
  function toast(msg, kind) {
    var t = el("div", { class: "toast " + (kind || "") }, msg);
    document.getElementById("toasts").appendChild(t);
    setTimeout(function () { t.remove(); }, kind === "bad" ? 6000 : 3500);
  }
  function statusBadge(s) {
    var map = { active: ["ok", "ativo"], expired: ["warn", "expirado"], revoked: ["bad", "revogado"] };
    var m = map[s] || ["", s];

  }
  function copy(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function () { toast("Copiado", "ok"); }, function () { toast("Não foi possível copiar", "bad"); });
    } else toast("Copie manualmente", "bad");
  }
  function qs(obj) {
    var p = [];
    Object.keys(obj).forEach(function (k) { if (obj[k] !== "" && obj[k] != null) p.push(encodeURIComponent(k) + "=" + encodeURIComponent(obj[k])); });
    return p.length ? "?" + p.join("&") : "";
  }

  function api(method, path, body) {
    var opt = { method: method, headers: { "X-Requested-With": "liveira-admin" }, credentials: "same-origin" };
    if (body !== undefined) { opt.headers["Content-Type"] = "application/json"; opt.body = JSON.stringify(body); }
    return fetch("/admin/api" + path, opt).then(function (r) {
      return r.json().catch(function () { return {}; }).then(function (d) {
        if (r.status === 401 && path !== "/login") { ME = null; renderLogin("Sessão expirada. Entre novamente."); throw new Error("Não autenticado"); }
        if (!r.ok) throw new Error(d.error || "Erro HTTP " + r.status);
        return d;
      });
    });
  }

  /* ───────── modal ───────── */
  function modal(title, body, actions) {
    var ov = el("div", { class: "ov" });
    var box = el("div", { class: "modal", role: "dialog", "aria-modal": "true" }, el("h2", null, title), body);
    var acts = el("div", { class: "acts" });
    function close() { ov.remove(); document.removeEventListener("keydown", onKey); }
    function onKey(e) { if (e.key === "Escape") close(); }
    (actions || [{ text: "Fechar" }]).forEach(function (a) {
      var b = el("button", { class: "btn " + (a.cls || ""), type: "button" }, a.text);
      b.addEventListener("click", function () {
        if (!a.onClick) return close();
        var r = a.onClick(close, b);
        if (r && r.then) { b.disabled = true; r.then(function (v) { b.disabled = false; if (v !== false) close(); }, function (e) { b.disabled = false; toast(e.message, "bad"); }); }
        else if (r !== false) close();
      });
      acts.appendChild(b);
    });
    box.appendChild(acts);
    ov.appendChild(box);
    ov.addEventListener("mousedown", function (e) { if (e.target === ov) close(); });
    document.addEventListener("keydown", onKey);
    document.body.appendChild(ov);
    var f = box.querySelector("input,textarea,select");
    if (f) setTimeout(function () { f.focus(); }, 30);
    return { close: close, box: box };
  }
  function confirmDlg(msg, okText, danger) {
    return new Promise(function (resolve) {
      var done = false;
      var m = modal("Confirmar", el("div", { class: "desc" }, msg), [
        { text: "Cancelar", onClick: function () { done = true; resolve(false); } },
        { text: okText || "Confirmar", cls: danger ? "d" : "p", onClick: function () { done = true; resolve(true); } },
      ]);
      var obs = new MutationObserver(function () { if (!document.body.contains(m.box)) { obs.disconnect(); if (!done) resolve(false); } });
      obs.observe(document.body, { childList: true });
    });
  }

      el("thead", null, el("tr", null, headers.map(function (h) { return el("th", { class: h.cls || null }, h.t || h); }))),
      el("tbody", null, rows)));

  function uploadXhr(pid, file, onProgress) {
    return new Promise(function (resolve, reject) {
      var x = new XMLHttpRequest();
      x.open("PUT", "/admin/api/products/" + encodeURIComponent(pid) + "/file");
      x.setRequestHeader("X-Requested-With", "liveira-admin");
      x.setRequestHeader("X-File-Name", encodeURIComponent(file.name));
      x.setRequestHeader("Content-Type", file.type || "application/octet-stream");
      x.upload.onprogress = function (e) { if (e.lengthComputable) onProgress(e.loaded / e.total); };
      x.onload = function () {
        var d = {}; try { d = JSON.parse(x.responseText); } catch (e) { /* ignore */ }
        if (x.status >= 200 && x.status < 300) resolve(d); else reject(new Error(d.error || "HTTP " + x.status));
      };
      x.onerror = function () { reject(new Error("Falha de rede no upload")); };
      x.send(file);
    });
  }

      var rows = d.entries.map(function (x) {
        var det = x.details_json || "";
        try { det = JSON.stringify(JSON.parse(det), null, 1); } catch (e) { /* raw */ }
        return el("tr", null, el("td", null, fmtDate(x.at)), el("td", { class: "mono" }, x.actor), el("td", null, el("span", { class: "badge" }, x.action)), el("td", null, el("div", { class: "pre" }, det)));

  /* ───────── Transmissão ───────── */
  VIEWS.broadcast = function (main) {
    return api("GET", "/broadcast/count").then(function (d) {
      var txt = el("textarea", { maxlength: 4000, placeholder: "Mensagem (texto simples)" });
      var bar = el("div"); var prog = el("div", { class: "progress" }, bar);
      var status = el("div", { class: "small muted" });
      var btn = el("button", { class: "btn p" }, "Enviar para todos");
      btn.addEventListener("click", function () {
        var text = txt.value.trim();
        if (!text) { toast("Escreva a mensagem", "bad"); return; }
        confirmDlg("Enviar esta mensagem para " + d.users + " usuário(s)? Esta ação não pode ser desfeita.\n\n" + text, "Enviar", true).then(function (ok) {
          if (!ok) return;
          btn.disabled = true; txt.disabled = true;
          var sent = 0, failed = 0;
          (function step(offset) {
            api("POST", "/broadcast", { text: text, offset: offset }).then(function (r) {
              sent += r.sent; failed += r.failed;
              bar.style.width = (r.total ? Math.round((r.next_offset / r.total) * 100) : 100) + "%";
              status.textContent = "Enviadas: " + sent + " · falhas: " + failed + " · " + r.next_offset + "/" + r.total;
              if (r.done) { toast("Transmissão concluída", "ok"); btn.disabled = false; txt.disabled = false; }
              else step(r.next_offset);
            }, function (e) { toast(e.message, "bad"); status.textContent += " — interrompido"; btn.disabled = false; txt.disabled = false; });
          })(0);
        });
      });
      clear(main).append(el("h1", null, "Transmissão"), el("div", { class: "card" },
        el("div", { class: "warnbox" }, "Envia uma mensagem do bot para todos os " + d.users + " usuário(s) cadastrados. Enviado em lotes de 25."),
        field("Mensagem", txt), prog, status, btn));
    });
  };

  /* ───────── boot ───────── */
  function boot() {
    fetch("/admin/api/me", { credentials: "same-origin" }).then(function (r) {
      if (r.status === 401) { renderLogin(); return null; }
      return r.json().then(function (d) { if (!r.ok) throw new Error(d.error || "Erro " + r.status); return d; });
    }).then(function (me) {
      if (!me) return;
      ME = me; PRODUCTS_CACHE = null;
      renderShell(); router();
    }).catch(function (e) { clear($app).appendChild(el("div", { class: "boot" }, "Erro: " + e.message)); });
  }
  boot();
})();
